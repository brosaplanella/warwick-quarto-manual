% Example TikZ figure. scripts/tikz.py turns it into unit-circle.pdf (PDF output)
% and unit-circle.svg (HTML output) before every render.
\documentclass[tikz, border=2pt]{standalone}
\input{palette.tex}  % module colours: brand-purple, brand-coral-60, accent, ...
\begin{document}
\begin{tikzpicture}[scale=2]
  \draw[->] (-1.2, 0) -- (1.3, 0);
  \draw[->] (0, -0.3) -- (0, 1.2);
  \draw[brand-purple, thick] (0, 0) circle (1);
  \coordinate (P) at (40:1);
  \fill[brand-coral-60] (0, 0) -- (1, 0) arc (0:40:1) -- cycle;
  \draw[brand-purple, thick] (0, 0) -- (P);
  \draw[dashed] (P) -- (P |- 0, 0);
  \draw[brand-purple, very thick] (P |- 0, 0) -- (1, 0)
    node[midway, below=4pt, fill=white, inner sep=1pt] {\small $1 - \cos x$};
  \node at (20:0.35) {\small $x$};
  \fill (P) circle (0.6pt);
\end{tikzpicture}
\end{document}
